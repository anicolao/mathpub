\begin{block}{Key Equation: Credit Utilization Ratio}
  \[
    \text{Credit Utilization Ratio} = \frac{\text{Total Balance Owed}}{\text{Total Credit Limit}} \times 100\%
  \]
\end{block}

\vspace{0.5em}

\begin{columns}[T]
  \begin{column}{0.5\textwidth}
    \textbf{Target Utilization:}
    \begin{itemize}
      \item There is no hard \textbf{30\% scoring cutoff}.
      \item Lower reported utilization is generally better, but its impact depends on the full credit profile.
    \end{itemize}
  \end{column}
  \begin{column}{0.5\textwidth}
    \textbf{Credit Report Contents:}
    \begin{itemize}
      \item Personal identification details.
      \item Account payment histories \& limits.
      \item Public records (e.g., bankruptcies).
      \item Credit inquiry logs (hard vs. soft).
    \end{itemize}
  \end{column}
\end{columns}
